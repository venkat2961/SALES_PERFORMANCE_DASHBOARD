-- Create these measures in Power BI (table names: Orders, Customers, Products, Date).
-- Use delivered orders for core sales KPIs. Returns/cancellations are excluded from the main KPIs.

Total Sales =
CALCULATE(SUM(Orders[Sales]), Orders[OrderStatus] = "Delivered")

Total Profit =
CALCULATE(SUM(Orders[Profit]), Orders[OrderStatus] = "Delivered")

Profit Margin % =
DIVIDE([Total Profit], [Total Sales], 0)

Total Orders =
CALCULATE(DISTINCTCOUNT(Orders[OrderID]), Orders[OrderStatus] = "Delivered")

Units Sold =
CALCULATE(SUM(Orders[Quantity]), Orders[OrderStatus] = "Delivered")

Unique Customers =
CALCULATE(DISTINCTCOUNT(Orders[CustomerID]), Orders[OrderStatus] = "Delivered")

Average Order Value =
DIVIDE([Total Sales], [Total Orders], 0)

Average Discount % =
CALCULATE(AVERAGE(Orders[DiscountPct]), Orders[OrderStatus] = "Delivered")

Returned Orders =
CALCULATE(DISTINCTCOUNT(Orders[OrderID]), Orders[OrderStatus] = "Returned")

Cancelled Orders =
CALCULATE(DISTINCTCOUNT(Orders[OrderID]), Orders[OrderStatus] = "Cancelled")

Sales Previous Month =
CALCULATE([Total Sales], DATEADD('Date'[Date], -1, MONTH))

Sales MoM Change % =
DIVIDE([Total Sales] - [Sales Previous Month], [Sales Previous Month], 0)

Sales Previous Year =
CALCULATE([Total Sales], SAMEPERIODLASTYEAR('Date'[Date]))

Sales YoY Change % =
DIVIDE([Total Sales] - [Sales Previous Year], [Sales Previous Year], 0)

-- Create a Date table:
Date = CALENDAR(MIN(Orders[OrderDate]), MAX(Orders[OrderDate]))
-- Then add calculated columns:
Year = YEAR('Date'[Date])
Month Number = MONTH('Date'[Date])
Month = FORMAT('Date'[Date], "MMM")
Year Month = FORMAT('Date'[Date], "YYYY-MM")
